# Регламент Сетевого Администратора Android ADB
1. Перед подключением убедись, что утилита `adb` доступна в системе (`adb version`).
2. Подключись к целевому IP: `adb connect <IP>:5555`.
3. Дождись подтверждения авторизации RSA-ключа на экране целевого устройства.
4. Проведи первичный аудит аппаратных ресурсов:
   - Процессор и архитектура: `adb shell getprop ro.product.cpu.abi`
   - Память и своп: `adb shell dumpsys meminfo` и `adb shell cat /proc/meminfo`
   - Дисковое пространство: `adb shell df -h /data`
5. ПРАВИЛО БЕЗОПАСНОСТИ: Перед отключением или удалением любых пакетов сформируй файл `rollback_adb.sh` с командами повторного включения (`pm enable <pkg>` / `cmd package install-existing <pkg>`).
6. Не пытайся рутировать устройство и не выполняй установку непроверенных неофициальных сборок.
7. После завершения сессии обязательно выполни `adb disconnect` и предупреди пользователя о необходимости отключения сетевой отладки в меню устройства.

Быстрый сбор данных о состоянии памяти, swap и диска Android-устройства через ADB в формате JSON.
python 
#!/usr/bin/env python3
import subprocess
import json
import re
import sys

def run_adb(cmd):
    res = subprocess.run(f"adb shell {cmd}", shell=True, capture_output=True, text=True)
    return res.stdout.strip()

def collect_telemetry():
    print("🔍 Сбор телеметрии с подключенного Android/TV устройства...")
    model = run_adb("getprop ro.product.model")
    abi = run_adb("getprop ro.product.cpu.abi")
    mem_info = run_adb("cat /proc/meminfo")
    
    total_mem = re.search(r"MemTotal:\s+(\d+)\s+kB", mem_info)
    free_mem = re.search(r"MemFree:\s+(\d+)\s+kB", mem_info)
    available_mem = re.search(r"MemAvailable:\s+(\d+)\s+kB", mem_info)
    swap_total = re.search(r"SwapTotal:\s+(\d+)\s+kB", mem_info)
    swap_free = re.search(r"SwapFree:\s+(\d+)\s+kB", mem_info)
    
    total_kb = int(total_mem.group(1)) if total_mem else 0
    avail_kb = int(available_mem.group(1)) if available_mem else 0
    sw_tot = int(swap_total.group(1)) if swap_total else 0
    sw_fr = int(swap_free.group(1)) if swap_free else 0
    sw_used = sw_tot - sw_fr
    
    disk_data = run_adb("df -h /data | tail -n 1").split()
    
    report = {
        "device_model": model,
        "cpu_architecture": abi,
        "ram_total_mb": round(total_kb / 1024, 1),
        "ram_available_mb": round(avail_kb / 1024, 1),
        "swap_used_mb": round(sw_used / 1024, 1),
        "storage_data_used": disk_data[2] if len(disk_data) > 2 else "N/A",
        "storage_data_free": disk_data[3] if len(disk_data) > 3 else "N/A"
    }
    print(json.dumps(report, indent=2, ensure_ascii=False))
    return report

if __name__ == "__main__":
    collect_telemetry()

#!/bin/bash
# Восстановление отключенных пакетов Amazon Fire TV
echo "🔄 Запуск отката изменений..."
PACKAGES=(
    "com.amazon.vizzini"
    "com.amazon.device.messaging"
    "com.amazon.device.sale.service"
    "com.amazon.whisperplay.contracts"
)

for pkg in "${PACKAGES[@]}"; do
    echo "Включение пакета: $pkg"
    adb shell pm enable "$pkg" 2>/dev/null || adb shell cmd package install-existing "$pkg"
done
echo "✅ Откат завершен. Перезагрузите устройство командой: adb reboot"
